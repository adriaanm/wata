# conversation-actions, phase 1: alice fills a conversation past the visible
# window, scrolls to the bottom of it, and redacts a message with F2.
#
# Thirteen clips is deliberate: twelve message rows fit between the header and
# the footer, so the thirteenth can only be reached by scrolling, and the
# scrolled window is what the second checkpoint pins.

family @bob:localhost

wait syncing 1 900
wait convs 1 900

# Each clip is one PTT press/release into the selected (family) conversation.
# `wait msgs` after each one keeps the timeline order deterministic — without
# it the sends would race and the rows could arrive interleaved.

advance 4
key ptt release
wait msgs 1 1500

# outrun the SENT flash before opening the conversation
advance 120
tap enter
advance 5
checkpoint conv-top

# twelve downs walk the selection from the oldest message to the newest, which
# sits one row past the window — the list scrolls by one.
tap down
tap down
tap down
tap down
tap down
tap down
tap down
tap down
tap down
tap down
tap down
tap down
advance 5
checkpoint conv-scrolled

# back up two rows, then holding red past BACK_HOLD_DELETE redacts the
# selected message (a red TAP would navigate back instead). The row count
# drops, which `wait` cannot see (it tests >=), so `waitmax` asserts it.
tap up
tap up
key back press
advance 30
key back release
waitmax msgs 12 1500
advance 5
checkpoint conv-deleted

# hold OK past OK_HOLD_FAVORITE (0.8s = 25 frames of 33ms) favorites the
# selected message: the row's star only appears once the server's
# net.wata.favorite state has come back through sync, which is what `favs`
# waits on (a short OK tap would play instead).
key enter press
advance 30
key enter release
wait favs 1 1500
advance 5
checkpoint conv-favorited
